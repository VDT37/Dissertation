## MONITORING 

# Monitoring the Jobs
squeue -u $USER

# Watch the queue live
watch -n 5 "squeue -u $USER"

# View the log (Recommended)
tail -f <jobname>-<job id>.out

# Check GPU usage while job is running
srun --jobid <job id> --pty nvidia-smi

# Inspect the memory usage of the job: Important field is MaxRSS, tells how much RAM it consuemed so that we can either increase the RAM or reduce the workers
sacct -j 42124194 --format=JobID,State,Elapsed,MaxRSS

sacct -j 42135052 \
  --format=JobID,JobName,State,Elapsed,MaxRSS,AveRSS,CPUTime,TotalCPU

# When job is running, open second terminal and watch memory usage
watch -n 30 'sstat -j 42128483.batch --format=MaxRSS'

## IF IT IS PENDING:
# Refresh the jobs queue every few seconds:
watch -n 10 "squeue -u $USER"

# Estimate when the job might start if it is pending
squeue -j <jobid> --start

# Check orchid availability
sinfo -p orchid

# Find recent jobs
sacct -u dv321

# Show all jobs from the last few days
sacct -u $USER -X \
  --starttime 2026-07-01 \
  --format=JobID,JobName,State,Elapsed,End \
  | grep COMPLETED

# Kill a submitted job:
scancel <jobid>
